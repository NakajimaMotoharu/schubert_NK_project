def crc64(binary):
    assert int.from_bytes(binary[6:8],'big')==69
    value=0
    for byte in binary:
        value^=byte
        for _ in range(8):value=(value>>1)^0xd800000000000000 if value&1 else value>>1
    return f'{value:016x}'

independent_crc={name[:-6]:crc64(path.read_bytes()) for name,path in zip(names,files)}
