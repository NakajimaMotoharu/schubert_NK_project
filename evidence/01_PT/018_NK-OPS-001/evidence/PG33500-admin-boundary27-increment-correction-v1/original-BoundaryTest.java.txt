  @Test
  void bootbound001CreatedFixedSqlBindingsAndIndependentPbkdf() throws Exception {
    var f = new Port();
    assertEquals(BootstrapAdminService.Outcome.CREATED, call(f));
    assertEquals(SQL, f.sql);
    assertEquals(
        List.of("R0", "P0", "R1", "P1", "L", "R2", "P2", "R3", "P3", "P4", "P5"), f.closes);
    var user = f.parameters.get(3);
    assertEquals(DISPLAY, user.get(1));
    assertEquals("new.admin", user.get(2));
    String stored = (String) user.get(3);
    assertTrue(stored.matches("\\{pbkdf2-v1}[0-9a-f]{96}"));
    byte[] bytes = HexFormat.of().parseHex(stored.substring(PASSWORD_PREFIX.length()));
    var key =
        new PBEKeySpec(
            "a".repeat(PASSWORD_LENGTH).toCharArray(), Arrays.copyOf(bytes, 16), 310_000, 256);
    try {
      String independent =
          PASSWORD_PREFIX
              + HexFormat.of().formatHex(Arrays.copyOf(bytes, 16))
              + HexFormat.of()
                  .formatHex(
                      SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256")
                          .generateSecret(key)
                          .getEncoded());
      assertEquals(independent, stored);
    } finally {
      key.clearPassword();
    }
    for (int index = 4; index <= 6; index++)
      assertEquals(NOW.atOffset(ZoneOffset.UTC), user.get(index));
    for (int index = 4; index <= 5; index++) {
      var values = f.parameters.get(index);
      assertEquals(NOW.atOffset(ZoneOffset.UTC), values.get(1));
      assertEquals(index == 4 ? "ACTION" : "SECURITY", values.get(2));
      assertEquals(MESSAGE, values.get(3));
      assertEquals(CORRELATION, values.get(4));
      assertEquals(42L, values.get(5));
      assertEquals(5, values.size());
    }
    assertEquals(6, user.size());
    f.finished();
  }

  @Test
  void bootbound016EveryCloseFailureAndTwrCompositeOrdering() {
    for (String site : CLOSE_SITES)
      for (Throwable failure :
          List.of(
              new SQLException("synthetic close"),
              new IllegalStateException("synthetic close"),
              new AssertionError("synthetic close"))) {
        var f = new Port();
        f.faults.put(site, failure);
        assertSame(failure, assertThrows(Throwable.class, () -> call(f)));
        f.finished();
      }
    var f = new Port();
    var primary = new SQLException("primary");
    var secondary = new IllegalStateException("secondary");
    f.faults.put("query0", primary);
    f.faults.put("closeP0", secondary);
    assertSame(primary, assertThrows(SQLException.class, () -> call(f)));
    assertArrayEquals(new Throwable[] {secondary}, primary.getSuppressed());
    f.finished();
    var same = new Port();
    var fatal = new AssertionError("same");
    same.faults.put("query0", fatal);
    same.faults.put("closeP0", fatal);
    assertSame(fatal, assertThrows(IllegalArgumentException.class, () -> call(same)).getCause());
    same.finished();
  }

  @Test
  void bootbound017FiniteSharedServiceExistingPath() throws Exception {
    var service = new BootstrapAdminService();
    var executor = Executors.newFixedThreadPool(WORKERS);
    try {
      var futures = new ArrayList<Future<Void>>();
      for (int index = 0; index < TASKS; index++)
        futures.add(
            executor.submit(
                () -> {
                  var f = new Port();
                  f.rows.get(2).set(0, true);
                  char[] raw = password();
                  assertEquals(
                      BootstrapAdminService.Outcome.ALREADY_EXISTS,
                      service.create(f.connection(), USER, raw, META));
                  assertArrayEquals(new char[raw.length], raw);
                  assertEquals(SQL.subList(0, 4), f.sql);
                  f.finished();
                  return null;
                }));
      for (var future : futures) assertNull(future.get(WAIT_SECONDS, TimeUnit.SECONDS));
    } finally {
      executor.shutdownNow();
      assertTrue(executor.awaitTermination(TERMINATION_SECONDS, TimeUnit.SECONDS));
    }
  }

    assertEquals(META, copy);
    assertFalse(META.equals(null));

    assertFalse(META.equals(new UserCommandMetadata(new UUID(9, 3), CORRELATION, NOW)));
    assertEquals("UserCommandMetadata[REDACTED]", META.toString());
    assertEquals(MESSAGE, META.messageId());
    assertEquals(CORRELATION, META.correlationId());
    assertEquals(NOW, META.at());
    assertEquals(1791166429, META.hashCode());
    for (int field = 0; field < 3; field++) {

          NullPointerException.class,

  private static char[] password() {
    return "a".repeat(PASSWORD_LENGTH).toCharArray();
  }

  private static BootstrapAdminService.Outcome call(Port port) throws SQLException {
    char[] raw = password();
    try {
      return new BootstrapAdminService().create(port.connection(), USER, raw, META);
    } finally {
      assertArrayEquals(new char[raw.length], raw);
    }
  }

  private static void invalid(Port port) {
    var failure = assertThrows(SQLException.class, () -> call(port));
    assertEquals(DIAGNOSTIC, failure.getMessage());
    assertNull(failure.getCause());
  }

  private static String canonical(String sql) {
    return sql.replaceAll("\\s+", " ").trim();
  }

  private static Object objectMethod(Object proxy, String method, Object[] args) {
    return switch (method) {
      case OBJECT_STRING -> "JdbcPort[REDACTED]";
      case OBJECT_HASH -> System.identityHashCode(proxy);
      case OBJECT_EQUALS -> proxy == args[0];
      default -> throw new AssertionError(method);
    };
  }

  private static final class Port {
    final List<List<Object>> rows = new ArrayList<>();
    final List<Map<Integer, Object>> parameters = new ArrayList<>();
    final List<String> sql = new ArrayList<>();
    final List<String> resources = new ArrayList<>();
    final List<String> acquired = new ArrayList<>();
    final List<String> closes = new ArrayList<>();
    final Map<String, Throwable> faults = new HashMap<>();
    final int[] cursors = {-1, -1, -1, -1};
    final boolean[] wasNull = new boolean[4];
    final int[] updated = {1, 1};
    boolean auto;
    boolean readonly;
    boolean nullPlain;
    int isolation = Connection.TRANSACTION_READ_COMMITTED;
    int prepared;
    int queries;
    int nullPrepared = -1;
    int nullResult = -1;

    Port() {
      rows.add(new ArrayList<>(Arrays.asList(true)));
      rows.add(new ArrayList<>(Arrays.asList("void")));
      rows.add(new ArrayList<>(Arrays.asList(false)));
      rows.add(new ArrayList<>(Arrays.asList(42L)));
    }

    void hit(String key) throws Throwable {
      var failure = faults.get(key);
      if (failure != null) throw failure;
    }

    void acquire(String id) {
      assertFalse(acquired.contains(id));
      acquired.add(id);
      resources.add(id);
    }

    void close(String id) throws Throwable {
      assertEquals(id, resources.removeLast());
      assertFalse(closes.contains(id));
      closes.add(id);
      hit(CLOSE_NAME + id);
    }

    void finished() {
      assertTrue(resources.isEmpty());
      assertEquals(acquired.size(), closes.size());
      assertTrue(closes.containsAll(acquired));
    }

    Connection connection() {
      return (Connection)
          Proxy.newProxyInstance(
              Connection.class.getClassLoader(),
              new Class<?>[] {Connection.class},
              (proxy, method, arguments) -> {
                var args = arguments == null ? new Object[0] : arguments;
                String name = method.getName();
                return switch (name) {
                  case "getAutoCommit" -> {
                    hit("auto");
                    yield auto;
                  }
                  case "isReadOnly" -> {
                    hit("readonly");
                    yield readonly;
                  }
                  case "getTransactionIsolation" -> {
                    hit("isolation");
                    yield isolation;
                  }
                  case "prepareStatement" -> {
                    int index = prepared++;
                    hit("prepare" + index);
                    parameters.add(new HashMap<>());
                    yield index == nullPrepared ? null : statement(index, (String) args[0]);
                  }
                  case "createStatement" -> {
                    hit("plain");
                    yield nullPlain ? null : plain();
                  }
                  case OBJECT_STRING, OBJECT_HASH, OBJECT_EQUALS -> objectMethod(proxy, name, args);
                  default -> throw new AssertionError("borrowed Connection method: " + name);
                };
              });
    }

    Statement plain() {
      acquire("L");
      return (Statement)
          Proxy.newProxyInstance(
              Statement.class.getClassLoader(),
              new Class<?>[] {Statement.class},
              (proxy, method, arguments) -> {
                var args = arguments == null ? new Object[0] : arguments;
                return switch (method.getName()) {
                  case "execute" -> {
                    hit("executeLock");
                    assertEquals(LOCK, args[0]);
                    sql.add((String) args[0]);
                    yield false;
                  }
                  case CLOSE_NAME -> {
                    close("L");
                    yield null;
                  }
                  case OBJECT_STRING, OBJECT_HASH, OBJECT_EQUALS ->
                      objectMethod(proxy, method.getName(), args);
                  default -> throw new AssertionError(method.getName());
                };
              });
    }

    PreparedStatement statement(int index, String query) {
      String id = "P" + index;
      acquire(id);
      return (PreparedStatement)
          Proxy.newProxyInstance(
              PreparedStatement.class.getClassLoader(),
              new Class<?>[] {PreparedStatement.class},
              (proxy, method, arguments) -> {
                var args = arguments == null ? new Object[0] : arguments;
                String name = method.getName();
                return switch (name) {
                  case "setString", "setObject", "setLong" -> {
                    hit(name + index + "." + args[0]);
                    assertNull(parameters.get(index).put((Integer) args[0], args[1]));
                    yield null;
                  }
                  case "executeQuery" -> {
                    hit("query" + index);
                    int q = queries++;
                    sql.add(canonical(query));
                    yield q == nullResult ? null : result(q);
                  }
                  case "executeUpdate" -> {
                    hit("update" + index);
                    assertTrue(index == 4 || index == 5);
                    sql.add(canonical(query));
                    yield updated[index - 4];
                  }
                  case CLOSE_NAME -> {
                    close(id);
                    yield null;
                  }
                  case OBJECT_STRING, OBJECT_HASH, OBJECT_EQUALS -> objectMethod(proxy, name, args);
                  default -> throw new AssertionError(name);
                };
              });
    }

    ResultSet result(int index) {
      String id = "R" + index;
      acquire(id);
      return (ResultSet)
          Proxy.newProxyInstance(
              ResultSet.class.getClassLoader(),
              new Class<?>[] {ResultSet.class},
              (proxy, method, arguments) -> {
                var args = arguments == null ? new Object[0] : arguments;
                return switch (method.getName()) {
                  case "next" -> {
                    hit("next" + index);
                    yield ++cursors[index] < rows.get(index).size();
                  }
                  case "getBoolean" -> {
                    hit("boolean" + index);
                    assertEquals(1, args[0]);
                    Object value = rows.get(index).get(cursors[index]);
                    wasNull[index] = value == null;
                    yield Boolean.TRUE.equals(value);
                  }
                  case "getLong" -> {
                    hit("long" + index);
                    assertEquals(1, args[0]);
                    Object value = rows.get(index).get(cursors[index]);
                    wasNull[index] = value == null;
                    yield value == null ? 0L : (Long) value;
                  }
                  case "wasNull" -> {
                    hit("wasnull" + index);
                    yield wasNull[index];
                  }
                  case CLOSE_NAME -> {
                    close(id);
                    yield null;
                  }
                  case OBJECT_STRING, OBJECT_HASH, OBJECT_EQUALS ->
                      objectMethod(proxy, method.getName(), args);
                  default -> throw new AssertionError(method.getName());
                };
              });
    }
  }
