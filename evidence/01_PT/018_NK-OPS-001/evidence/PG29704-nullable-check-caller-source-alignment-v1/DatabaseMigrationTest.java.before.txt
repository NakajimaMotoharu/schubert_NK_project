  @BeforeAll
  static void startPostgreSqlAndMigrate() throws Exception {
    pgRoot = locatePostgreSqlRoot();
    Path target = MODULE.resolve("target");
    Files.createDirectories(target);
    clusterRoot = Files.createTempDirectory(target, "postgresql-18-");
    dataDirectory = clusterRoot.resolve("data");
    port = availablePort();

    run(
        pgBinary("initdb"),
        "-D",
        dataDirectory.toString(),
        "-L",
        pgRoot.resolve("share/postgresql/18").toString(),
        "--no-locale",
        "--encoding=UTF8",
        "--auth=trust",
        "--username=" + ADMIN);
    run(
        pgBinary("pg_ctl"),
        "-D",
        dataDirectory.toString(),
        "-l",
        clusterRoot.resolve("postgresql.log").toString(),
        "-o",
        serverOptions(),
        "start");

    run(
        pgBinary(FIXTURE_CREATEDB_AF6714D1),

    flyway = configuredFlyway(jdbcUrl, true, FIXTURE_CLASSPATH_DB_MIGRATION_0053F0D8);
    firstMigration = flyway.migrate();

  @AfterAll

      run(pgBinary("pg_ctl"), "-D", dataDirectory.toString(), "stop", "-m", "fast");
    }

  @Test
  @Order(1)
  @DisplayName("PT-DB-001 V001-V012 migrate, validate and idempotent rerun")
  void migrationsAreVersionedValidatedAndIdempotent() throws SQLException {
    assertEquals(MIGRATION_VERSIONS.size(), firstMigration.migrationsExecuted);
    assertTrue(flyway.validateWithResult().validationSuccessful);
    assertEquals(0, flyway.migrate().migrationsExecuted);
    List<String> applied =
        Arrays.stream(flyway.info().applied())
            .map(MigrationInfo::getVersion)
            .filter(Objects::nonNull)
            .map(version -> version.getVersion())
            .toList();
    assertEquals(MIGRATION_VERSIONS, applied);
    assertEquals(
        0L,
        queryLong(
            ADMIN,
            "select count(*) from schubert_meta.flyway_schema_history where success = false"));
    assertEquals(
        "nk_owner",
        queryString(
            ADMIN,
            "select pg_get_userbyid(relowner) from pg_class where oid='schubert_meta.flyway_schema_history'::regclass"));
  }

  private static void strictCheck(
      Connection c, String table, String fields, boolean accepted, String constraint)
      throws SQLException {
    Savepoint point = c.setSavepoint();
    try {
      String sql = "update schubert." + table + " set " + fields;
      if (accepted) {
        try (Statement statement = c.createStatement()) {
          assertEquals(1, statement.executeUpdate(sql));
        }
      } else {
        SQLException error = assertThrows(SQLException.class, () -> execute(c, sql));
        assertEquals("23514", error.getSQLState());
        assertTrue(error instanceof PSQLException);
        var details = ((PSQLException) error).getServerErrorMessage();
        assertNotNull(details);
        assertEquals(constraint, details.getConstraint());
      }
    } finally {
      c.rollback(point);
      c.releaseSavepoint(point);
    }
  }

        pgRoot.resolve("lib/x86_64-linux-gnu") + ":" + pgRoot.resolve("lib/postgresql/18/lib");

    assertEquals(0, process.exitValue(), "command failed: " + String.join(" ", command));

    try (Statement statement = connection.createStatement();
        ResultSet result = statement.executeQuery(sql)) {

  private static Map<String, Set<String>> queryGroupedSet(String sql) throws SQLException {
    try (Connection connection = connection(ADMIN);
        Statement statement = connection.createStatement();
        ResultSet result = statement.executeQuery(sql)) {
      Map<String, Set<String>> values = new LinkedHashMap<>();
      while (result.next()) {
        values
            .computeIfAbsent(result.getString(1), ignored -> new LinkedHashSet<>())
            .add(result.getString(2));
      }
      return values;
    }
  }

  private static void copyMigrationResources(Path target) throws Exception {
    for (int version = 1; version <= 5; version++) {
      String prefix = "V%03d__".formatted(version);
      Path sourceDirectory = MODULE.resolve("src/main/resources/db/migration");
      Path source;
      try (var files = Files.list(sourceDirectory)) {
        source =
            files
                .filter(path -> path.getFileName().toString().startsWith(prefix))
                .findFirst()
                .orElseThrow();
      }
      Files.copy(
          source,
          target.resolve(source.getFileName()),
          java.nio.file.StandardCopyOption.REPLACE_EXISTING);
    }
  }
